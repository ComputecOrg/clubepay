	})

		t.Fatalf("failed to start postgres container: %v", err)
	}

	t.Cleanup(func() { container.Terminate(ctx) })

	host, _ := container.Host(ctx)
	port, _ := container.MappedPort(ctx, "5432")
	dsn := fmt.Sprintf("postgres://test:test@%s:%s/clubepay_test?sslmode=disable", host, port.Port())

	// Run migrations
	_, filename, _, _ := runtime.Caller(0)
	migrationsPath := filepath.Join(filepath.Dir(filename), "..", "..", "migrations")
	migrationsPath = filepath.ToSlash(migrationsPath)

	m, err := migrate.New("file://"+migrationsPath, dsn)
	if err != nil {
		t.Fatalf("failed to create migrate instance: %v", err)
	}
	if err := m.Up(); err != nil && err != migrate.ErrNoChange {
		t.Fatalf("failed to run migrations: %v", err)
	}

	pool, err := pgxpool.New(ctx, dsn)
	if err != nil {
		t.Fatalf("failed to connect to test db: %v", err)
	}

	t.Cleanup(func() { pool.Close() })

	return pool
}

func MigrationsDir() string {
	_, filename, _, _ := runtime.Caller(0)
	return filepath.Join(filepath.Dir(filename), "..", "..", "migrations")

// backend/internal/domain/auth.go
package domain

import (
	"fmt"
	"time"

	"github.com/golang-jwt/jwt/v5"
	"golang.org/x/crypto/bcrypt"
)

type JWTClaims struct {
	UserID int64  `json:"user_id"`
	Role   string `json:"role"`
	jwt.RegisteredClaims
}

func HashPassword(password string) (string, error) {
	hash, err := bcrypt.GenerateFromPassword([]byte(password), bcrypt.DefaultCost)
	if err != nil {
		return "", fmt.Errorf("hash password: %w", err)
	}
	return string(hash), nil
}

func CheckPassword(password, hash string) bool {
	return bcrypt.CompareHashAndPassword([]byte(hash), []byte(password)) == nil
}

func GenerateJWT(userID int64, role, secret string, expiry time.Duration) (string, error) {
	claims := JWTClaims{
		UserID: userID,
		Role:   role,
		RegisteredClaims: jwt.RegisteredClaims{
			ExpiresAt: jwt.NewNumericDate(time.Now().Add(expiry)),
			IssuedAt:  jwt.NewNumericDate(time.Now()),
		},
	}
	token := jwt.NewWithClaims(jwt.SigningMethodHS256, claims)
	return token.SignedString([]byte(secret))
}

func ParseJWT(tokenStr, secret string) (*JWTClaims, error) {
	token, err := jwt.ParseWithClaims(tokenStr, &JWTClaims{}, func(t *jwt.Token) (interface{}, error) {
		if _, ok := t.Method.(*jwt.SigningMethodHMAC); !ok {
			return nil, fmt.Errorf("unexpected signing method: %v", t.Header["alg"])
		}
		return []byte(secret), nil
	})
	if err != nil {
		return nil, fmt.Errorf("parse jwt: %w", err)
	}
	claims, ok := token.Claims.(*JWTClaims)
	if !ok || !token.Valid {
		return nil, ErrInvalidToken
	}
	return claims, nil
}

package handler

import (
	"encoding/json"

	"github.com/clubepay/backend/internal/config"
	"github.com/clubepay/backend/internal/psp"

	Queries *repository.Queries
	Config  *config.Config
	PSP     psp.PSP
}

	return &Handler{Queries: q, Config: cfg, PSP: p}
}

func writeJSON(w http.ResponseWriter, status int, data interface{}) {
	w.Header().Set("Content-Type", "application/json")
	w.WriteHeader(status)
	if err := json.NewEncoder(w).Encode(data); err != nil {
		slog.Error("failed to encode response", "error", err)
	}
}

func writeError(w http.ResponseWriter, status int, message string) {
	writeJSON(w, status, map[string]string{"error": message})
}

func readJSON(r *http.Request, dst interface{}) error {
	defer r.Body.Close()
	decoder := json.NewDecoder(r.Body)
	decoder.DisallowUnknownFields()
	return decoder.Decode(dst)
}

// backend/internal/middleware/auth.go
package middleware

import (
	"context"
	"encoding/json"
	"net/http"
	"strings"

	"github.com/clubepay/backend/internal/domain"
)

type contextKey string

const (
	userIDKey contextKey = "userID"
	roleKey   contextKey = "role"
)

func Auth(jwtSecret string) func(http.Handler) http.Handler {
	return func(next http.Handler) http.Handler {
		return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
			authHeader := r.Header.Get("Authorization")
			if authHeader == "" {
				writeErr(w, http.StatusUnauthorized, "missing authorization header")
				return
			}

			parts := strings.SplitN(authHeader, " ", 2)
			if len(parts) != 2 || parts[0] != "Bearer" {
				writeErr(w, http.StatusUnauthorized, "invalid authorization format")
				return
			}

			claims, err := domain.ParseJWT(parts[1], jwtSecret)
			if err != nil {
				writeErr(w, http.StatusUnauthorized, "invalid token")
				return
			}

			ctx := context.WithValue(r.Context(), userIDKey, claims.UserID)
			ctx = context.WithValue(ctx, roleKey, claims.Role)
			next.ServeHTTP(w, r.WithContext(ctx))
		})
	}
}

func RequireRole(role string) func(http.Handler) http.Handler {
	return func(next http.Handler) http.Handler {
		return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
			if RoleFromContext(r.Context()) != role {
				writeErr(w, http.StatusForbidden, "forbidden")
				return
			}
			next.ServeHTTP(w, r)
		})
	}
}

func UserIDFromContext(ctx context.Context) int64 {
	id, _ := ctx.Value(userIDKey).(int64)
	return id
}

func RoleFromContext(ctx context.Context) string {
	role, _ := ctx.Value(roleKey).(string)
	return role
}

func writeErr(w http.ResponseWriter, status int, msg string) {
	w.Header().Set("Content-Type", "application/json")
	w.WriteHeader(status)
	json.NewEncoder(w).Encode(map[string]string{"error": msg})
}

// backend/internal/middleware/cors.go
package middleware

import "net/http"

func CORS(next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		w.Header().Set("Access-Control-Allow-Origin", "*")
		w.Header().Set("Access-Control-Allow-Methods", "GET, POST, PUT, DELETE, OPTIONS")
		w.Header().Set("Access-Control-Allow-Headers", "Content-Type, Authorization")

		if r.Method == "OPTIONS" {
			w.WriteHeader(http.StatusOK)
			return
		}

		next.ServeHTTP(w, r)
	})
}

// backend/internal/handler/auth.go
package handler

import (
	"net/http"
	"strings"
	"time"
	"unicode"

	"github.com/clubepay/backend/internal/domain"
	"github.com/clubepay/backend/internal/repository"
)

type registerOwnerRequest struct {
	Email        string `json:"email"`
	Password     string `json:"password"`
	Name         string `json:"name"`
	BusinessName string `json:"business_name"`
	Segment      string `json:"segment"`
}

func (h *Handler) RegisterOwner(w http.ResponseWriter, r *http.Request) {
	var req registerOwnerRequest
	if err := readJSON(r, &req); err != nil {
		writeError(w, http.StatusBadRequest, "invalid request body")
		return
	}

	if req.Email == "" || req.Password == "" || req.Name == "" || req.BusinessName == "" {
		writeError(w, http.StatusBadRequest, "email, password, name, and business_name are required")
		return
	}

	if len(req.Password) < 8 {
		writeError(w, http.StatusBadRequest, "password must be at least 8 characters")
		return
	}

	hash, err := domain.HashPassword(req.Password)
	if err != nil {
		writeError(w, http.StatusInternalServerError, "internal error")
		return
	}

	user, err := h.Queries.CreateUser(r.Context(), repository.CreateUserParams{
		Email:        req.Email,
		PasswordHash: hash,
		Name:         req.Name,
		Role:         domain.RoleOwner,
	})
	if err != nil {
		if strings.Contains(err.Error(), "duplicate key") {
			writeError(w, http.StatusConflict, "email already exists")
			return
		}
		writeError(w, http.StatusInternalServerError, "failed to create user")
		return
	}

	segment := req.Segment
	if segment == "" {
		segment = "cafeteria"
	}

	slug := generateSlug(req.BusinessName)

	biz, err := h.Queries.CreateBusiness(r.Context(), repository.CreateBusinessParams{
		OwnerID: user.ID,
		Name:    req.BusinessName,
		Slug:    slug,
		Segment: segment,
	})
	if err != nil {
		writeError(w, http.StatusInternalServerError, "failed to create business")
		return
	}

	token, err := domain.GenerateJWT(user.ID, user.Role, h.Config.JWTSecret, 24*time.Hour)
	if err != nil {
		writeError(w, http.StatusInternalServerError, "failed to generate token")
		return
	}

	writeJSON(w, http.StatusCreated, map[string]interface{}{
		"token":    token,
		"user":     user,
		"business": biz,
	})
}

func generateSlug(name string) string {
	slug := strings.ToLower(name)
	var b strings.Builder
	for _, r := range slug {
		if unicode.IsLetter(r) || unicode.IsDigit(r) {
			b.WriteRune(r)
		} else if r == ' ' || r == '-' {
			b.WriteRune('-')
		}
	}
	result := b.String()
	// Remove double dashes
	for strings.Contains(result, "--") {
		result = strings.ReplaceAll(result, "--", "-")
	}
	return strings.Trim(result, "-")
}

const freeTierPlanLimit = 1

type createPlanRequest struct {
	Name        string `json:"name"`
	Description string `json:"description"`
	PriceCents  int64  `json:"price_cents"`
	LimitType   string `json:"limit_type"`
	LimitCount  int32  `json:"limit_count"`

func (h *Handler) CreatePlan(w http.ResponseWriter, r *http.Request) {
	userID := middleware.UserIDFromContext(r.Context())

	biz, err := h.Queries.GetBusinessByOwnerID(r.Context(), userID)
	if err != nil {
		writeError(w, http.StatusNotFound, "business not found")
		return
	}

	var req createPlanRequest
	if err := readJSON(r, &req); err != nil {
		writeError(w, http.StatusBadRequest, "invalid request body")
		return
	}

	if req.Name == "" || req.PriceCents <= 0 || req.LimitType == "" || req.LimitCount <= 0 {
		writeError(w, http.StatusBadRequest, "name, price_cents, limit_type, and limit_count are required")
		return
	}

	if req.LimitType != "daily" && req.LimitType != "monthly" {
		writeError(w, http.StatusBadRequest, "limit_type must be 'daily' or 'monthly'")
		return
	}

	// Free tier: max 1 plan
	count, err := h.Queries.CountPlansByBusinessID(r.Context(), biz.ID)
	if err != nil {
		writeError(w, http.StatusInternalServerError, "failed to check plan count")
		return
	}
	if count >= freeTierPlanLimit {
		writeError(w, http.StatusForbidden, "plan limit reached for free tier")
		return
	}

	plan, err := h.Queries.CreatePlan(r.Context(), repository.CreatePlanParams{
		BusinessID:  biz.ID,
		Name:        req.Name,
		Description: pgText(req.Description),
		PriceCents:  req.PriceCents,
		LimitType:   req.LimitType,
		LimitCount:  req.LimitCount,
	})
	if err != nil {
		writeError(w, http.StatusInternalServerError, "failed to create plan")
		return
	}

	writeJSON(w, http.StatusCreated, plan)

// backend/internal/handler/subscription.go
package handler

import (
	"net/http"
	"strconv"
	"time"

	"github.com/go-chi/chi/v5"
	"github.com/clubepay/backend/internal/domain"
	"github.com/clubepay/backend/internal/middleware"
	"github.com/clubepay/backend/internal/psp"
	"github.com/clubepay/backend/internal/repository"
)

const freeTierSubscriberLimit = 15

type subscribeRequest struct {
	PlanID int64  `json:"plan_id"`
	CPF    string `json:"cpf"`
}

func (h *Handler) Subscribe(w http.ResponseWriter, r *http.Request) {
	userID := middleware.UserIDFromContext(r.Context())

	var req subscribeRequest
	if err := readJSON(r, &req); err != nil {
		writeError(w, http.StatusBadRequest, "invalid request body")
		return
	}

	if req.PlanID == 0 {
		writeError(w, http.StatusBadRequest, "plan_id is required")
		return
	}

	plan, err := h.Queries.GetPlanByID(r.Context(), req.PlanID)
	if err != nil {
		writeError(w, http.StatusNotFound, "plan not found")
		return
	}

	if !plan.Active {
		writeError(w, http.StatusBadRequest, "plan is not active")
		return
	}

	// Check already subscribed
	_, err = h.Queries.GetActiveSubscription(r.Context(), repository.GetActiveSubscriptionParams{
		SubscriberID: userID,
		BusinessID:   plan.BusinessID,
	})
	if err == nil {
		writeError(w, http.StatusConflict, "already subscribed to this business")
		return
	}

	// Free tier check
	count, err := h.Queries.CountActiveSubscriptionsByBusinessID(r.Context(), plan.BusinessID)
	if err != nil {
		writeError(w, http.StatusInternalServerError, "failed to check subscriber count")
		return
	}
	if count >= freeTierSubscriberLimit {
		writeError(w, http.StatusForbidden, "subscriber limit reached for free tier")
		return
	}

	// Get user for PSP customer
	user, err := h.Queries.GetUserByID(r.Context(), userID)
	if err != nil {
		writeError(w, http.StatusInternalServerError, "failed to get user")
		return
	}

	// Create PSP customer + subscription
	customer, err := h.PSP.CreateCustomer(r.Context(), psp.CreateCustomerRequest{
		Name:  user.Name,
		Email: user.Email,
		CPF:   req.CPF,
	})
	if err != nil {
		writeError(w, http.StatusInternalServerError, "failed to create PSP customer")
		return
	}

type validateUsageRequest struct {
	BusinessSlug string `json:"business_slug"`

func (h *Handler) ValidateUsage(w http.ResponseWriter, r *http.Request) {
	userID := middleware.UserIDFromContext(r.Context())

	var req validateUsageRequest
	if err := readJSON(r, &req); err != nil {
		writeError(w, http.StatusBadRequest, "invalid request body")
		return
	}

	biz, err := h.Queries.GetBusinessBySlug(r.Context(), req.BusinessSlug)
	if err != nil {
		writeError(w, http.StatusNotFound, "business not found")
		return
	}

	sub, err := h.Queries.GetActiveSubscription(r.Context(), repository.GetActiveSubscriptionParams{
		SubscriberID: userID,
		BusinessID:   biz.ID,
	})
	if err != nil {
		writeError(w, http.StatusForbidden, "no active subscription")
		return
	}

	if sub.Status == domain.SubscriptionStatusBlocked || sub.Status == domain.SubscriptionStatusCancelled {
		writeError(w, http.StatusForbidden, "subscription is not active")
		return
	}

	plan, err := h.Queries.GetPlanByID(r.Context(), sub.PlanID)
	if err != nil {
		writeError(w, http.StatusInternalServerError, "failed to get plan")
		return
	}

	// Check usage limit
	now := time.Now()
	var usageCount int64
	if plan.LimitType == domain.LimitTypeDaily {
		start, end := domain.DailyRange(now)
		usageCount, err = h.Queries.CountDailyUsage(r.Context(), repository.CountDailyUsageParams{
			SubscriptionID: sub.ID,
			ValidatedAt:    pgTimestamptz(start),
			ValidatedAt_2:  pgTimestamptz(end),
		})
	} else {
		start, end := domain.MonthlyRange(now)
		usageCount, err = h.Queries.CountMonthlyUsage(r.Context(), repository.CountMonthlyUsageParams{
			SubscriptionID: sub.ID,
			ValidatedAt:    pgTimestamptz(start),
			ValidatedAt_2:  pgTimestamptz(end),
		})
	}
	if err != nil {
		writeError(w, http.StatusInternalServerError, "failed to check usage")
		return
	}

	if err := domain.CheckUsageLimit(plan.LimitType, int(plan.LimitCount), usageCount); err != nil {
		writeError(w, http.StatusForbidden, "usage limit reached for this period")
		return
	}

	// Record usage
	usage, err := h.Queries.CreateUsage(r.Context(), sub.ID)
	if err != nil {
		writeError(w, http.StatusInternalServerError, "failed to record usage")
		return
	}

	writeJSON(w, http.StatusOK, map[string]interface{}{
		"status":       "validated",
		"usage_id":     usage.ID,
		"used":         usageCount + 1,
		"limit":        plan.LimitCount,
		"limit_type":   plan.LimitType,
		"plan_name":    plan.Name,
	})
}

```go
// backend/internal/handler/webhook.go
package handler

import (
	"io"
	"encoding/json"
	"log/slog"
	"net/http"
	"time"

	"github.com/clubepay/backend/internal/domain"
	"github.com/clubepay/backend/internal/repository"
)

type webhookPayload struct {
	Event   string         `json:"event"`
	Payment webhookPayment `json:"payment"`
}

type webhookPayment struct {
	Subscription string `json:"subscription"`
	Status       string `json:"status"`
}

func (h *Handler) PSPWebhook(w http.ResponseWriter, r *http.Request) {
	body, err := io.ReadAll(r.Body)
	if err != nil {
		writeError(w, http.StatusBadRequest, "failed to read body")
		return
	}
	defer r.Body.Close()

	// Validate HMAC (simplified — use h.PSP.ValidateWebhook in production)
	signature := r.Header.Get("asaas-access-token")
	if h.Config.AsaasWebhookSecret != "" && !h.PSP.ValidateWebhook(body, signature) {
		writeError(w, http.StatusUnauthorized, "invalid webhook signature")
		return
	}

	var payload webhookPayload
	if err := json.Unmarshal(body, &payload); err != nil {
		writeError(w, http.StatusBadRequest, "invalid payload")
		return
	}

	if payload.Payment.Subscription == "" {
		w.WriteHeader(http.StatusOK) // Ignore non-subscription events
		return
	}

	sub, err := h.Queries.GetSubscriptionByPSPID(r.Context(), pgText(payload.Payment.Subscription))
	if err != nil {
		slog.Warn("webhook for unknown subscription", "psp_id", payload.Payment.Subscription)
		w.WriteHeader(http.StatusOK) // Don't retry
		return
	}

	switch payload.Event {
	case "PAYMENT_CONFIRMED", "PAYMENT_RECEIVED":
		// Reactivate if was in grace/blocked
		if sub.Status != domain.SubscriptionStatusActive {
			h.Queries.UpdateSubscriptionStatus(r.Context(), repository.UpdateSubscriptionStatusParams{
				ID:     sub.ID,
				Status: domain.SubscriptionStatusActive,
			})
		}

	case "PAYMENT_OVERDUE":
		// Start 3-day grace period
		graceDL := time.Now().AddDate(0, 0, 3)
		h.Queries.UpdateSubscriptionGrace(r.Context(), repository.UpdateSubscriptionGraceParams{
			ID:            sub.ID,
			GraceDeadline: pgTimestamptz(graceDL),
		})

	case "PAYMENT_DELETED", "PAYMENT_REFUNDED":
		h.Queries.CancelSubscription(r.Context(), sub.ID)
	}

	w.WriteHeader(http.StatusOK)
}
```

// backend/internal/handler/cron.go
package handler

import (
	"log/slog"
	"net/http"

	"github.com/clubepay/backend/internal/domain"
	"github.com/clubepay/backend/internal/repository"
)

func (h *Handler) Reconcile(w http.ResponseWriter, r *http.Request) {
	secret := r.Header.Get("X-Cron-Secret")
	if secret != h.Config.CronSecret {
		writeError(w, http.StatusUnauthorized, "invalid cron secret")
		return
	}

	ctx := r.Context()
	var blocked, synced int

	// 1. Block expired grace subscriptions
	expired, err := h.Queries.ListGraceExpiredSubscriptions(ctx)
	if err != nil {
		slog.Error("failed to list expired grace subscriptions", "error", err)
	} else {
		for _, sub := range expired {
			if err := h.Queries.UpdateSubscriptionStatus(ctx, repository.UpdateSubscriptionStatusParams{
				ID:     sub.ID,
				Status: domain.SubscriptionStatusBlocked,
			}); err != nil {
				slog.Error("failed to block subscription", "id", sub.ID, "error", err)
			} else {
				blocked++
			}
		}
	}

	// 2. Sync pending subscriptions with PSP
	pending, err := h.Queries.ListPendingSubscriptions(ctx)
	if err != nil {
		slog.Error("failed to list pending subscriptions", "error", err)
	} else {
		for _, sub := range pending {
			pspSub, err := h.PSP.GetSubscription(ctx, sub.PspSubscriptionID)
			if err != nil {
				slog.Error("failed to get PSP subscription", "id", sub.PspSubscriptionID, "error", err)
				continue
			}

			var newStatus string
			switch pspSub.Status {
			case "ACTIVE":
				newStatus = domain.SubscriptionStatusActive
			case "INACTIVE", "EXPIRED":
				newStatus = domain.SubscriptionStatusCancelled
			default:
				continue
			}

			if newStatus != sub.Status {
				h.Queries.UpdateSubscriptionStatus(ctx, repository.UpdateSubscriptionStatusParams{
					ID:     sub.ID,
					Status: newStatus,
				})
				synced++
			}
		}
	}

	writeJSON(w, http.StatusOK, map[string]interface{}{
		"blocked": blocked,
		"synced":  synced,
	})
}

package psp

import (
	"bytes"
	"context"
	"crypto/hmac"
	"crypto/sha256"
	"encoding/hex"
	"encoding/json"
	"fmt"
	"io"
	"net/http"

type Asaas struct {
	baseURL       string
	apiKey        string
	webhookSecret string
	client        *http.Client
}

func NewAsaas(baseURL, apiKey, webhookSecret string) *Asaas {
	return &Asaas{
		baseURL:       baseURL,
		apiKey:        apiKey,
		webhookSecret: webhookSecret,
		client:        &http.Client{},
	}
}

func (a *Asaas) CreateCustomer(ctx context.Context, req CreateCustomerRequest) (*Customer, error) {
	body := map[string]string{
		"name":       req.Name,
		"email":      req.Email,
		"cpfCnpj":    req.CPF,
		"mobilePhone": req.Phone,
	}

	var result struct {
		ID string `json:"id"`
	}
	if err := a.post(ctx, "/customers", body, &result); err != nil {
		return nil, err
	}

	return &Customer{ID: result.ID}, nil
}

func (a *Asaas) CreateSubscription(ctx context.Context, req CreateSubscriptionRequest) (*Subscription, error) {
	body := map[string]interface{}{
		"customer":    req.CustomerID,
		"billingType": "PIX",
		"value":       float64(req.PriceCents) / 100,
		"cycle":       req.Cycle,
		"description": req.Description,
	}

	var result struct {
		ID     string `json:"id"`
		Status string `json:"status"`
	}
	if err := a.post(ctx, "/subscriptions", body, &result); err != nil {
		return nil, err
	}

	return &Subscription{ID: result.ID, Status: result.Status}, nil
}

func (a *Asaas) CancelSubscription(ctx context.Context, subscriptionID string) error {
	req, err := http.NewRequestWithContext(ctx, "DELETE", a.baseURL+"/subscriptions/"+subscriptionID, nil)
	if err != nil {
		return err
	}
	req.Header.Set("access_token", a.apiKey)

	resp, err := a.client.Do(req)
	if err != nil {
		return err
	}
	defer resp.Body.Close()

	if resp.StatusCode >= 400 {
		return fmt.Errorf("asaas cancel failed: status %d", resp.StatusCode)
	}
	return nil
}

func (a *Asaas) GetSubscription(ctx context.Context, subscriptionID string) (*Subscription, error) {
	var result struct {
		ID     string `json:"id"`
		Status string `json:"status"`
	}
	if err := a.get(ctx, "/subscriptions/"+subscriptionID, &result); err != nil {
		return nil, err
	}
	return &Subscription{ID: result.ID, Status: result.Status}, nil
}

func (a *Asaas) GetPayments(ctx context.Context, subscriptionID string) ([]Payment, error) {
	var result struct {
		Data []struct {
			ID     string `json:"id"`
			Status string `json:"status"`
		} `json:"data"`
	}
	if err := a.get(ctx, "/subscriptions/"+subscriptionID+"/payments", &result); err != nil {
		return nil, err
	}

	payments := make([]Payment, len(result.Data))
	for i, p := range result.Data {
		payments[i] = Payment{ID: p.ID, Status: p.Status}
	}
	return payments, nil
}

func (a *Asaas) ValidateWebhook(payload []byte, signature string) bool {
	if a.webhookSecret == "" {
		return true
	}
	mac := hmac.New(sha256.New, []byte(a.webhookSecret))
	mac.Write(payload)
	expected := hex.EncodeToString(mac.Sum(nil))
	return hmac.Equal([]byte(expected), []byte(signature))
}

func (a *Asaas) post(ctx context.Context, path string, body interface{}, result interface{}) error {
	b, _ := json.Marshal(body)
	req, err := http.NewRequestWithContext(ctx, "POST", a.baseURL+path, bytes.NewReader(b))
	if err != nil {
		return err
	}
	req.Header.Set("Content-Type", "application/json")
	req.Header.Set("access_token", a.apiKey)

	resp, err := a.client.Do(req)
	if err != nil {
		return err
	}
	defer resp.Body.Close()

	if resp.StatusCode >= 400 {
		respBody, _ := io.ReadAll(resp.Body)
		return fmt.Errorf("asaas error %d: %s", resp.StatusCode, string(respBody))
	}

	return json.NewDecoder(resp.Body).Decode(result)
}

func (a *Asaas) get(ctx context.Context, path string, result interface{}) error {
	req, err := http.NewRequestWithContext(ctx, "GET", a.baseURL+path, nil)
	if err != nil {
		return err
	}
	req.Header.Set("access_token", a.apiKey)

	resp, err := a.client.Do(req)
	if err != nil {
		return err
	}
	defer resp.Body.Close()

	return json.NewDecoder(resp.Body).Decode(result)
}
